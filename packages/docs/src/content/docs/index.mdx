<CardGrid>
	<Card title="Owns the Stripe integration" icon="puzzle">
		Customers, plans, one-time payments, subscriptions, invoices and webhook

	</Card>
	<Card title="Idempotent webhooks" icon="random">

	</Card>
	<Card title="Resilient Stripe calls" icon="approve-check">

	</Card>
	<Card title="Typed client" icon="seti:typescript">

		the service's OpenAPI spec.
	</Card>
